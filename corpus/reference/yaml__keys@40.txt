# a flow collection. Prettier strips the explicit `?` marker, which is
# anonymous, so that rewrite lives in normalisation.yaml instead.

# Flow-collection keys first: a comment sitting between two plain pairs
# and a flow-collection key is a prettier syntax error ("Map comment with
# trailing content"), even though tree-sitter-yaml accepts it.

{ k: v }: map-as-key
