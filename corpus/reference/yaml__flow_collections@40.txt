# Flow collections are the construct prettier actually reflows. Width 80
# keeps the medium ones flat; width 40 expands them one element per line
# and adds a trailing comma. Block form is a different file.

empty_seq: [] # empty
empty_map: {}
short_seq: [1, 2, 3]
short_map: { a: 1, b: 2 }
# Fits at 80, explodes at 40.
medium_seq:
  [alpha, bravo, charlie, delta, echo]
medium_map:
  {
    host: localhost,
    port: 8080,
    role: primary,
  }
# Still flat at 80; one item per line at 40.
long_seq:
  [
    alpha,
    bravo,
    charlie,
    delta,
    echo,
    foxtrot,
    golf,
    hotel,
  ]
long_map:
  {
    host: alpha.example.com,
    port: 8080,
    region: us-east-1,
  }
# A hand-broken flow that fits is collapsed, because it has no comment.
already_broken: [keep, me, open]
# A source trailing comma is dropped when the collection stays flat.
trailing_comma: [1, 2, 3]
mixed:
  [1, two, 3.0, true, { k: 1 }, [8, 9]]
