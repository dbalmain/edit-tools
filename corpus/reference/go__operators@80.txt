package main

func arithmetic(a, b, c, d, e int) int {
	return a + b*c + d/e
}

func combined(a, b, c, d, e int) int {
	return a*b + c/d - e%7
}

func shifts(a int) int {
	return a<<4 + a>>1
}

func bitwise(a, b, c int) int {
	return a | b&c
}

func andNot(a, b int) int {
	return a &^ b
}

func comparison(a, b, c, d int) bool {
	return a+b < c+d && a != 0 // mixed precedence comparison
}

func unary(p *int) int {
	return -*p
}

func address(x int) *int {
	return &x
}

func not(ok bool) bool {
	return !ok
}
